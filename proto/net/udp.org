- unicast :: single =src= to single =dst= (TCP and UDP)
- broadcast :: many =src= to many =dst= on a single network (UDP)
- anycast :: single =src= to one of many =dst= (TCP [kinda] and UDP)
- multicast :: many =src= to many =dst= (UDP and RTP*)
