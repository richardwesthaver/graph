  __attribute__((force_align_arg_pointer))
  void _start() {
    /* call main(), etc */

    /* exit system call */
    // 32-bit
    // asm("movl $1,%eax;xorl %ebx,%ebx;int  $0x80");
    // 64-bit

    __builtin_unreachable();  // tell the compiler to make sure side effects are done before the asm statement
  }
